# Network isolation proof — docker compose (compose.yaml), 2026-09-27T15:21Z
# Docker 29.7.2, Compose 5.5.1

## 1. The frontend cannot reach the database (must fail)
$ docker compose exec -T frontend ping -c 1 -W 2 database
ping: bad address 'database'
(exit 1)

$ docker compose exec -T frontend nc -z -w 2 database 5432
nc: bad address 'database'
(exit 1)

$ docker compose exec -T frontend nc -z -w 2 postgres 5432
nc: bad address 'postgres'
(exit 1)

$ docker compose exec -T frontend nc -z -w 2 redis 6379
nc: bad address 'redis'
(exit 1)

## 2. Positive controls: the same checks from places that should succeed
$ docker compose exec -T frontend nc -z -w 2 backend 8000
(exit 0)

$ docker compose exec -T backend python -c import socket; socket.create_connection(('database', 5432), timeout=2); print('backend -> database:5432 open')
backend -> database:5432 open
(exit 0)

$ docker compose exec -T backend python -c import socket; socket.create_connection(('redis', 6379), timeout=2); print('backend -> redis:6379 open')
backend -> redis:6379 open
(exit 0)

## 3. Nothing on the internal network can reach the internet (must fail)
$ docker compose exec -T postgres wget -q -T 3 -O /dev/null http://example.com
wget: bad address 'example.com'
(exit 1)

$ docker compose exec -T redis sh -c nc -z -w 3 1.1.1.1 443
(exit 1)

## 4. Which networks each service joined
frontend: civicpulse_edge 
backend: civicpulse_edge civicpulse_internal 
postgres: civicpulse_internal 
redis: civicpulse_internal 
## 5. Not just DNS: the frontend cannot reach Postgres by IP either (must fail)
$ docker compose exec -T frontend nc -z -w 2 172.18.0.3 5432   # postgres's address on civicpulse_internal
(exit 1)

$ docker compose exec -T backend python -c "...create_connection(('172.18.0.3', 5432))"   # positive control
open
(exit 0)
